Singleton.java
// Singleton — один экземпляр на JVM
// Лучший способ: enum (потокобезопасен, защищён от рефлексии и десериализации)
enum AppConfig {
    INSTANCE;
    private final String host = System.getenv("HOST");
    public String getHost() { return host; }
}
AppConfig.INSTANCE.getHost();

// Double-checked locking (если enum не подходит)
class DBPool {
    private static volatile DBPool instance; // volatile обязателен!

    private DBPool() {}

    public static DBPool getInstance() {
        if (instance == null) {
            synchronized (DBPool.class) {
                if (instance == null) {  // повторная проверка под локом
                    instance = new DBPool();
                }
            }
        }
        return instance;
    }
}
Factory.java
// Factory Method — делегирует создание подклассам
interface Logger {
    void log(String message);
}
class ConsoleLogger implements Logger {
    public void log(String m) { System.out.println("[CONSOLE] " + m); }
}
class FileLogger implements Logger {
    public void log(String m) { /* запись в файл */ }
}

class LoggerFactory {
    public static Logger create(String type) {
        return switch (type) {
            case "console" -> new ConsoleLogger();
            case "file"    -> new FileLogger();
            default -> throw new IllegalArgumentException(type);
        };
    }
}

Builder.java
// Builder — пошаговое создание сложных объектов
class HttpRequest {
    private final String url;
    private final String method;
    private final Map<String, String> headers;
    private final String body;

    private HttpRequest(Builder b) {
        this.url = b.url; this.method = b.method;
        this.headers = b.headers; this.body = b.body;
    }

    public static class Builder {
        private final String url;
        private String method = "GET";
        private Map<String, String> headers = new HashMap<>();
        private String body;

        public Builder(String url) { this.url = url; }
        public Builder method(String m) { this.method = m; return this; }
        public Builder header(String k, String v) { headers.put(k, v); return this; }
        public Builder body(String b) { this.body = b; return this; }
        public HttpRequest build() { return new HttpRequest(this); }
    }
}

// Использование
HttpRequest req = new HttpRequest.Builder("https://api.example.com/users")
    .method("POST")
    .header("Content-Type", "application/json")
    .body("{\"name\":\"Alice\"}")
    .build();

    public Map<String, Object> parse(String json) {

Decorator.java
// Decorator — добавляет поведение без изменения оригинала
interface Coffee { double getCost(); String getDescription(); }

class SimpleCoffee implements Coffee {
    public double getCost() { return 1.0; }
    public String getDescription() { return "Coffee"; }
}

abstract class CoffeeDecorator implements Coffee {
    protected Coffee coffee;
    CoffeeDecorator(Coffee c) { this.coffee = c; }
}

class MilkDecorator extends CoffeeDecorator {
    MilkDecorator(Coffee c) { super(c); }
    public double getCost() { return coffee.getCost() + 0.5; }
    public String getDescription() { return coffee.getDescription() + ", Milk"; }
}

Coffee c = new MilkDecorator(new MilkDecorator(new SimpleCoffee()));
// Coffee, Milk, Milk — стоимость 2.0

Proxy.java
// Proxy — контролирует доступ к объекту (логирование, кэширование, security)
interface Service { String getData(String key); }
class RealService implements Service {
    public String getData(String key) { return "data:" + key; }
}

// Статический прокси
class LoggingProxy implements Service {
    private final Service service;
    LoggingProxy(Service s) { this.service = s; }

    public String getData(String key) {
        System.out.println("Calling getData(" + key + ")");
        String result = service.getData(key);
        System.out.println("Result: " + result);
        return result;
    }
}

// Динамический прокси (JDK Proxy)
Service proxy = (Service) Proxy.newProxyInstance(
    Service.class.getClassLoader(),
    new Class[]{Service.class},
    (p, method, args) -> {
        System.out.println("Before: " + method.getName());
        Object result = method.invoke(new RealService(), args);
        System.out.println("After: " + method.getName());
        return result;
    }
);

9.3Поведенческие: Strategy, Observer, Command

Strategy.java
// Strategy — семейство алгоритмов, взаимозаменяемых в runtime
@FunctionalInterface
interface SortStrategy<T> { void sort(List<T> list); }

class Sorter<T> {
    private SortStrategy<T> strategy;
    Sorter(SortStrategy<T> s) { this.strategy = s; }
    void setStrategy(SortStrategy<T> s) { this.strategy = s; }
    void sort(List<T> list) { strategy.sort(list); }
}

// Лямбды отлично работают как стратегии
Sorter<Integer> sorter = new Sorter<>(Collections::sort);
sorter.setStrategy(list -> list.sort(Comparator.reverseOrder()));

// В JDK: Comparator, Executor, Runnable сами являются стратегиями
Observer.java
// Observer — подписка на события
interface EventListener<T> { void onEvent(T event); }

class EventBus<T> {
    private final List<EventListener<T>> listeners = new CopyOnWriteArrayList<>();

    void subscribe(EventListener<T> l) { listeners.add(l); }
    void unsubscribe(EventListener<T> l) { listeners.remove(l); }
    void publish(T event) { listeners.forEach(l -> l.onEvent(event)); }
}

EventBus<String> bus = new EventBus<>();
bus.subscribe(e -> System.out.println("Received: " + e));
bus.publish("Hello"); // Received: Hello

Command.java
// Command — инкапсулирует запрос как объект (undo, очередь задач)
interface Command { void execute(); void undo(); }

class TextEditor {
    private StringBuilder text = new StringBuilder();
    private Deque<Command> history = new ArrayDeque<>();

    void executeCommand(Command cmd) {
        cmd.execute();
        history.push(cmd);
    }

    void undo() {
        if (!history.isEmpty()) history.pop().undo();
    }

    class InsertCommand implements Command {
        private final String text;
        InsertCommand(String t) { this.text = t; }
        public void execute() { TextEditor.this.text.append(text); }
        public void undo()    { TextEditor.this.text.delete(
            TextEditor.this.text.length() - text.length(),
            TextEditor.this.text.length()); }
    }
}
